# 以下是关联关系表的 测试代码


-- ========================================
-- 1. 创建客户表 t_customer
-- ========================================
CREATE TABLE `t_customer` (
    `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '客户ID，主键自增',
    `customer_name` VARCHAR(255) NOT NULL COMMENT '客户姓名',
    `phone` VARCHAR(255) NOT NULL COMMENT '电话号码',
    PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='客户表';

-- ========================================
-- 2. 创建订单表 t_order
-- ========================================
CREATE TABLE `t_order` (
    `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '订单ID，主键自增',
    `address` VARCHAR(255) NOT NULL COMMENT '收货地址',
    `amount` DECIMAL(10,2) NOT NULL COMMENT '订单金额（两位小数）',
    `customer_id` BIGINT NOT NULL COMMENT '所属客户ID（外键）',
    PRIMARY KEY (`id`),
    KEY `idx_customer_id` (`customer_id`),
    CONSTRAINT `fk_order_customer` FOREIGN KEY (`customer_id`) REFERENCES `t_customer` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='订单表';
